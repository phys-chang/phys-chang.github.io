\newcommand{\ultraheading}[3]{
  \fill[BlueTint] (90,#1) circle (35bp);
  \node[text=Navy,font=\fontsize{40}{42}\selectfont\bfseries,inner sep=0bp]
    at (90,#1) {#2};
  \node[text=Navy,anchor=west,font=\fontsize{40}{42}\selectfont\bfseries,inner sep=0bp]
    at (152,#1) {#3};
}

\newcommand{\ultrarim}[2]{

    \pgfmathsetmacro{\rimouta}{36*sqrt(max(0,1-pow((\rimya-435)/70,2)))}
    \pgfmathsetmacro{\rimoutb}{36*sqrt(max(0,1-pow((\rimyb-435)/70,2)))}
    \pgfmathsetmacro{\rimina}{28*sqrt(max(0,1-pow((\rimya-435)/62,2)))}
    \pgfmathsetmacro{\riminb}{28*sqrt(max(0,1-pow((\rimyb-435)/62,2)))}
    \pgfmathtruncatemacro{\rimshade}{18+54*\n/79}
    \fill[Blue!\rimshade!white]
      ({#1-\rimouta},\rimya) -- ({#1-\rimina},\rimya)
      -- ({#1-\riminb},\rimyb) -- ({#1-\rimoutb},\rimyb) -- cycle;
    \fill[Blue!\rimshade!white]
      ({#1+\rimina},\rimya) -- ({#1+\rimouta},\rimya)
      -- ({#1+\rimoutb},\rimyb) -- ({#1+\riminb},\rimyb) -- cycle;

% Shaded cylindrical lattice: open axial ends, periodic circumference.
% Keep the same projection and circumferential defect coordinates.
\newcommand{\ultracylinder}[1]{
  \begin{scope}[shift={(#1,0)}]
    \foreach \n in {0,...,7} {
      \pgfmathsetmacro{\shadowrx}{226-7*\n}
      \pgfmathsetmacro{\shadowry}{15-\n}
      \fill[Navy,opacity=.012]
        (437,519) ellipse (\shadowrx bp and \shadowry bp);
    }
    % A soft highlight above center and a deeper blue underside create volume.
    % Direct curved bands remain portable to SVG and PDF renderers.
    \foreach \n in {0,...,99} {
      \pgfmathsetmacro{\surfaceya}{365+140*\n/100}
      \pgfmathsetmacro{\surfaceyb}{min(505,\surfaceya+1.8)}
      \pgfmathsetmacro{\surfacexa}{36*sqrt(max(0,1-pow((\surfaceya-435)/70,2)))}
      \pgfmathsetmacro{\surfacexb}{36*sqrt(max(0,1-pow((\surfaceyb-435)/70,2)))}
      \pgfmathsetmacro{\surfaceangle}{asin(-1+2*(\n+.5)/100)}
      \pgfmathtruncatemacro{\surfaceshade}{15+70*(1-max(0,cos(\surfaceangle+30)))}
      \fill[Blue!\surfaceshade!white]
        ({244+\surfacexa},\surfaceya) -- ({594+\surfacexa},\surfaceya)
        -- ({594+\surfacexb},\surfaceyb) -- ({244+\surfacexb},\surfaceyb) -- cycle;
    }
    \draw[white,opacity=.4,line width=5bp,line cap=round]
      (285,399) -- (607,399);
    \foreach \x in {244,314,384,454,524,594} {
      \draw[BlueTint!70!white,line width=1.8bp]
        plot[domain=-90:90,samples=35,variable=\a]
          ({\x+36*cos(\a)},{435+70*sin(\a)});
    }
    \foreach \a in {-60,-30,0,30,60} {
      \draw[BlueTint!65!white,line width=1.6bp]
        ({244+36*cos(\a)},{435+70*sin(\a)})
